% Table I of the paper (the Fig. 2 ladder numbers) as a standalone figure, so it
% can be combined with the Fig. 2 artwork in a vector editor.
%
%   latex fig2_ladder_table.tex && dvips -E -o fig2_ladder_table.eps fig2_ladder_table.dvi
%   epstopdf --outfile=fig2_ladder_table.pdf fig2_ladder_table.eps
%   gs -q -dNOPAUSE -dBATCH -dSAFER -sDEVICE=png16m -r600 -dEPSCrop \
%      -dTextAlphaBits=4 -dGraphicsAlphaBits=4 \
%      -sOutputFile=fig2_ladder_table.png fig2_ladder_table.eps
%
% The PNG is 600 dpi (693x1992 px), twice the 300 dpi the matplotlib figures
% export at, so it survives being scaled up next to them.
%
% dvips -E (not pdflatex) keeps the glyphs as Type 1 text so Illustrator can
% still select them -- the same reason the other figures avoid alpha fills.
% Body only, no caption: the caption text stays in main.tex.
% Font and size mirror main.tex's `table` block: ieeeconf sets \rmdefault to
% ptm (Times) at 10 pt and the table is \footnotesize.
%
% PORTRAIT form (user's call, 2026-09-14): one block per panel, quantity left
% and value right, so the table is taller than it is wide and sits beside the
% path solutions rather than under them. The wide 5-column form that matches
% main.tex's Table I verbatim is in this file's history.
\documentclass[10pt]{article}
\usepackage[T1]{fontenc}
\renewcommand{\rmdefault}{ptm}
\usepackage{amsmath}
\usepackage{booktabs}
\usepackage[active,tightpage]{preview}
\PreviewEnvironment{tabular}
\setlength\PreviewBorder{1pt}
\pagestyle{empty}

% One panel block: label row, then the four quantities.
\newcommand{\panelblock}[5]{%
  \multicolumn{2}{@{}l@{}}{#1}\\[1pt]
  \quad Level          & #2\,\%  \\
  \quad Bound $\bar u$ & #3\,m   \\
  \quad Length         & #4\,m   \\
  \quad Terminal $\sigma$ & #5\,m \\
}

\begin{document}
\footnotesize
\setlength{\tabcolsep}{6pt}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}lr@{}}
  \toprule
  \panelblock{(a)}{100}{13.75}{1741}{9.73}
  \midrule
  \panelblock{(b)}{70}{9.62}{1761}{8.89}
  \midrule
  \panelblock{(c)}{50}{6.87}{1822}{6.66}
  \midrule
  \panelblock{(d)}{30}{4.12}{1890}{3.91}
  \bottomrule
\end{tabular}
\end{document}
